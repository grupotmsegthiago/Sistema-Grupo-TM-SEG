/**
 * FONTE ÚNICA DA VERDADE - CÁLCULOS FINANCEIROS
 * Este código unifica Relatório, Termômetro, Dashboard e E-mail.
 */

export const calculateMissionRevenue = (mission: any) => {
  // 1. Filtro de exclusão: Refused não gera receita
  if (mission.status === 'REFUSED') return 0;

  // 2. Cálculo da Receita (Valor base + Pedágio)
  // Prioriza valor aprovado/salvo e cai para estimativa se estiver zerado
  const baseRevenue = mission.revenue_value || mission.estimated_value || 0;
  const toll = mission.toll_value || 0;

  return baseRevenue + toll;
};

export const getCanonicalDateRange = (period: 'month' | 'week' | 'today') => {
  const start = new Date();
  const end = new Date();
  end.setHours(23, 59, 59, 999); // Sempre até o fim de hoje

  if (period === 'month') {
    start.setDate(1);
    start.setHours(0, 0, 0, 0);
  } else if (period === 'week') {
    const day = start.getDay(); // 0 (Dom) a 6 (Sáb)
    start.setDate(start.getDate() - day);
    start.setHours(0, 0, 0, 0);
  } else if (period === 'today') {
    start.setHours(0, 0, 0, 0);
  }

  return { start, end };
};

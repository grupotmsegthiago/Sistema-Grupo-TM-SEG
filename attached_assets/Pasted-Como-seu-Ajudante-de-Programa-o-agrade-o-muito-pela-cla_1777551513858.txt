TypeScript
{/* Insira este bloco ao lado do seu botão "MESMA OS" no header */}
<div className="flex items-center gap-2">
  {/* Caso 1: A OS é Filha */}
  {mission.is_same_os && mission.parent_mission_id && (
    <span className="bg-blue-500/20 text-blue-300 border border-blue-500/50 px-2 py-0.5 rounded text-[10px] font-bold uppercase tracking-wider">
      Filha de: {mission.parent_mission_id}
    </span>
  )}

  {/* Caso 2: A OS é Mãe (Verifica se há filhas no array de vinculadas) */}
  {!mission.is_same_os && linkedMissions.filter(m => m.parent_mission_id === mission.id).length > 0 && (
    <span className="bg-amber-500/20 text-amber-300 border border-amber-500/50 px-2 py-0.5 rounded text-[10px] font-bold uppercase tracking-wider">
      Mãe ({linkedMissions.filter(m => m.parent_mission_id === mission.id).length} vinculadas)
    </span>
  )}
</div>
